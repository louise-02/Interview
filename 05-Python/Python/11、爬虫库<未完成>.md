# 1、requests

requests 是 Python 最常用的 HTTP 客户端库。

底层基于 urllib3，比 urllib.request 更简单、好用。

官方文档：https://cn.python-requests.org/zh_CN/latest/user/quickstart.html

安装：pip install requests -i https://pypi.douban.com/simple

## 基本使用

```python
import requests

url = "https://httpbin.org/get"

response = requests.get(url)

# 获取网页源码（字符串）
print(response.text)

# 获取响应编码
print(response.encoding)

# 实际请求地址（含参数）
print(response.url)

# 响应的字节数据（二进制）
print(response.content)

# HTTP 状态码
print(response.status_code)

# 响应头（字典）
print(response.headers)

# JSON 响应（最常用）
data = response.json()     # 自动反序列化为 dict / list
print(data)
print(type(data))          # dict

# 是否请求成功的判断
print(response.ok)         # True / False
# True  -> status_code < 400
# False -> status_code >= 400

# 主动抛出 HTTP 异常（强烈推荐）
try:
    response.raise_for_status()
except requests.exceptions.HTTPError as e:
    print("HTTP 请求失败:", e)
    
# 请求耗时（性能分析）
print(response.elapsed)            # timedelta 对象
print(response.elapsed.total_seconds(), "秒")

# 获取单个响应头
print(response.headers.get("Content-Type"))
print(response.headers.get("Server"))

# 编码问题（中文乱码）编码问题（中文乱码）
response.encoding = "utf-8"
print(response.text)

# 二进制下载（图片 / 文件）
img_url = "https://httpbin.org/image/png"
resp = requests.get(img_url)

with open("test.png", "wb") as f:
    f.write(resp.content)
    
# 超时控制（非常重要）
response = requests.get(
    url,
    timeout=5   # 秒，防止一直卡住
)
```

## GET 请求

```python
import requests

url = "https://httpbin.org/get"

# GET 参数会拼接到 URL 后面
params = {
    "username": "louise",
    "age": 18
}

headers = {
    "User-Agent": "Mozilla/5.0"
}

response = requests.get(url, params=params, headers=headers)

# httpbin 会把你传的参数原样返回
print(response.text)
```

## POST 请求

```python
import requests
import json

url = "https://httpbin.org/post"

headers = {
    "User-Agent": "Mozilla/5.0",
    "Content-Type": "application/json;charset=UTF-8"
}

data = {
    "username": "louise",
    "password": "123456"
}

# 转成 JSON + 二进制
json_data = json.dumps(data).encode("utf-8")

response = requests.post(
    url,
    headers=headers,
    data=json_data
)

print(response.text)



# 更推荐的写法
url = "https://httpbin.org/post"

data = {
    "username": "louise",
    "password": "123456"
}

response = requests.post(url, json=data)

print(response.json())
```

## 代理 proxies

```python
import requests

url = "https://httpbin.org/ip"

# HTTP / HTTPS 代理
proxies = {
    "http": "http://202.55.5.209:8090",
    "https": "http://202.55.5.209:8090"
}

# ⚠️ 代理不一定可用，报错属于正常
response = requests.get(url, proxies=proxies, timeout=5)

print(response.text)
```

## session 会话

```python
# session 用于：
# 1. 自动保存 cookie
# 2. 多次请求复用连接
# 3. 模拟登录态

import requests

session = requests.session()

# 第一次请求
r1 = session.get("https://httpbin.org/cookies/set", params={"token": "abc123"})
print(r1.text)

# 第二次请求（会带上 cookie）
r2 = session.get("https://httpbin.org/cookies")
print(r2.text)
```

## 超时 timeout

```python
import requests

url = "https://httpbin.org/delay/3"

try:
    response = requests.get(url, timeout=2)
    print(response.text)
except requests.exceptions.Timeout:
    print("请求超时")
```

## 异常处理

```python
import requests

url = "https://httpbin.org/status/404"

try:
    response = requests.get(url)
    response.raise_for_status()   # 非 2xx 状态码会抛异常
except requests.exceptions.HTTPError as e:
    print("HTTP 错误:", e)
except requests.exceptions.RequestException as e:
    print("请求异常:", e)
```

# 2、XPath（HTML 解析）

XPath：一种在 HTML / XML 文档中定位元素的路径语言，常用于爬虫中从网页结构里“精准取值”。

安装库：pip install lxml

安装浏览器插件：XPath Helper

```python
from lxml import etree
import requests

# ===== 方式一：解析本地 HTML 文件 =====
# url = "test.html"
# tree = etree.parse(url, encoding="utf-8")

# ===== 方式二：解析网页响应内容（最常用） =====
url = "https://httpbin.org/html"
response = requests.get(url)
# HTML() 用于解析字符串形式的 HTML
tree = etree.HTML(response.text)

# -------------------------
# 2. XPath 基础用法
# -------------------------

# XPath 返回结果：
# - text() → 字符串列表
# - 属性 → 字符串列表
# - 节点 → Element 对象列表

# 示例：获取所有 h1 标签文本
result = tree.xpath("//h1/text()")
print(result)
```

## XPath 语法总结

```python
# ---------- 路径查询 ----------
# /     ：从当前节点找「直接子节点」
# //    ：从当前节点找「所有子孙节点（不考虑层级）」

# 查找所有 div
# //div

# 查找 body 下的直接 div
# /html/body/div


# ---------- 谓词查询 ----------
# [条件] 用于筛选节点

# 含有 id 属性的 div
# //div[@id]

# id 等于 maincon 的 div
# //div[@id="maincon"]


# ---------- 属性查询 ----------
# 使用 @属性名

# 获取 class 属性
# //div/@class


# ---------- 模糊匹配 ----------
# contains()  → 包含
# starts-with() → 以…开头

# id 中包含 he
# //div[contains(@id, "he")]

# id 以 he 开头
# //div[starts-with(@id, "he")]


# ---------- 内容查询 ----------
# text() 获取文本内容

# 获取 h1 标签中的文本
# //div/h1/text()


# ---------- 逻辑运算 ----------
# and / or

# 同时满足 id 和 class
# //div[@id="head" and @class="s_down"]


# ---------- 多路径查询 ----------
# | 表示“或”

# 同时获取 title 和 price
# //title | //price
```

## 实战示例

```python
from lxml import etree
import requests

url = "https://httpbin.org/html"
response = requests.get(url)

# HTML() 用于解析字符串形式的 HTML
tree = etree.HTML(response.text)

# 获取所有链接文本
links = tree.xpath("//a/text()")
print(links)

# 获取所有链接地址
hrefs = tree.xpath("//a/@href")
print(hrefs)

# 获取第一个 h1（索引从 1 开始）
h1 = tree.xpath("//h1[1]/text()")
print(h1)
```

## 注意事项

```python
# ❌ 忘了 text()，拿到的是节点对象
# //h1

# ❌ 用索引 0
# //h1[0]   # XPath 索引从 1 开始

# ❌ HTML 未规范，parse() 报错
# 👉 使用 HTML() 更安全

# ❌ 返回的一定是数组 需要注意
```

# 3、Scrapy

Scrapy：是一个为了爬取网站数据，提取结构性数据而编写的应用框架。可以应用在包括数据挖掘，信息处理或存储历史数据等一系列的程序中。

安装库：pip install scrapy -i https://mirrors.aliyun.com/pypi/simple/

## 基础操作

```python
# 测试站点：https://httpbin.org/html

# 创建项目
# scrapy startproject 项目名
# 注意：
# 1. 不能以数字开头
# 2. 不能有中文
# 3. 建议全小写
# 示例：
scrapy startproject myspider

# 进入 spiders 目录
cd myspider/myspider/spiders

# 创建爬虫
# scrapy genspider 爬虫名 要爬的网页
# 示例：
scrapy genspider test httpbin.org

# 运行爬虫
# scrapy crawl 爬虫名
# 示例：
scrapy crawl test

#
```
