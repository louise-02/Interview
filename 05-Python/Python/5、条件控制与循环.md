# 1、条件控制

```python
# =========================
# 1. 条件控制 if / elif / else
# =========================
# 根据条件执行不同代码块
# 条件表达式返回 True / False

score = 85

if score >= 90:
    print("优秀")
elif score >= 60:
    print("及格")
else:
    print("不及格")


# ⚠️ 注意：
# 1）elif 可以有多个
# 2）else 可省略
# 3）Python 用缩进而不是大括号控制代码块
```

# 2、for 循环

```python
# =========================
# 2. for 循环
# =========================
# 用于遍历“可迭代对象”（list / tuple / str / set / dict / generator）

lst = ['a', 'b', 'c']

for item in lst:
    print(item)

# for + else
# ⚠️ else 在“循环未被 break 打断时”才会执行

for i in range(3):
    print(i)
else:
    print("循环正常结束")
```

# 3、while 循环

```python
# =========================
# 3. while 循环
# =========================
# 当条件为 True 时反复执行

count = 0
while count < 3:
    print(count)
    count += 1


# while + else
# 同样：未被 break 打断才会执行 else

count = 0
while count < 3:
    print(count)
    count += 1
else:
    print("while 正常结束")
```

# 4、range() 函数

```python
# =========================
# 4. range() 函数
# =========================
# range(start, stop, step)
# ⚠️ stop 不包含在内
# 返回的是“可迭代对象”（惰性），不是列表

# range(stop)
for i in range(5):
    print(i)       # 0 1 2 3 4

# range(start, stop, step)
for i in range(0, 10, 3):
    print(i)       # 0 3 6 9


# range + 索引（不推荐，了解即可）
a = ['Google', 'Baidu', 'Runoob', 'Taobao', 'QQ']

for i in range(len(a)):
    print(i, a[i])

# ✅ 推荐写法
for i, v in enumerate(a):
    print(i, v)
```

# 5、迭代器 iterator

```python
# =========================
# 5. 迭代器 iterator
# =========================
# 任何“可迭代对象”都可以通过 iter() 生成迭代器
# 迭代器通过 next() 逐个取值
# 取完后抛出 StopIteration

lst = [1, 2, 3, 4]
it = iter(lst)

for x in it:
    print(x, end=" ")
# 1 2 3 4


# 手动使用 next()
lst = [1, 2, 3, 4]
it = iter(lst)

while True:
    try:
        print(next(it))
    except StopIteration:
        break
```

# 6、补充总结

```python
# =========================
# 补充总结（很重要）
# =========================
# for 本质上就是不断调用 next()
# for 循环内部自动处理 StopIteration
# 迭代器是“一次性”的，用完就没了
# range / enumerate / zip / generator 都是迭代器或可迭代对象


# =========================
# 1️⃣ range —— 可迭代对象（惰性生成）
# =========================
r = range(3)
print(r)                # range(0, 3)

for i in r:
    print(i)
# 0
# 1
# 2

# 可以手动迭代
it = iter(r)
print(next(it))         # 0
print(next(it))         # 1


# =========================
# 2️⃣ enumerate —— 可迭代对象（包装序列）
# =========================
lst = ['a', 'b', 'c']

e = enumerate(lst)
print(e)                # <enumerate object ...>

for i, v in e:
    print(i, v)
# 0 a
# 1 b
# 2 c


# =========================
# 3️⃣ zip —— 可迭代对象（并行遍历）
# =========================
names = ['Tom', 'Bob', 'Alice']
scores = [90, 85, 88]

z = zip(names, scores)
print(z)                # <zip object ...>

for name, score in z:
    print(name, score)
# Tom 90
# Bob 85
# Alice 88

# ⚠️ zip 以最短序列为准
print(list(zip([1, 2, 3], ['a', 'b'])))
# [(1, 'a'), (2, 'b')]


# =========================
# 4️⃣ generator —— 迭代器（惰性计算）
# =========================
gen = (x * x for x in range(3))
print(gen)              # <generator object ...>

print(next(gen))        # 0
print(next(gen))        # 1
print(next(gen))        # 4
# print(next(gen))      # ❌ StopIteration

# 生成器只能用一次
gen = (x for x in range(3))
print(list(gen))        # [0, 1, 2]
print(list(gen))        # []


# =========================
# 对比总结（非常关键）
# =========================
# range / enumerate / zip
#   → 返回的是“可迭代对象”
#   → 可用于 for / iter / next
#
# generator
#   → 返回的是“迭代器”
#   → 只能遍历一次
#   → 常用于大数据、流式处理
```
