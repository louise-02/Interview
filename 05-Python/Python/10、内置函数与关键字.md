# 1、常用内置函数

```python
# =========================
# Python 常用内置函数（类型 / 对象 / 反射相关）
# =========================
# 这些都属于 Python 的 built-in functions（内置函数）
# 无需 import，随时可用


# =========================
# 1. type()
# =========================
# 返回对象的“真实类型”
# ⚠️ 不考虑继承关系

a = True
print(type(a))          # <class 'bool'>
print(type(a) is int)   # False（bool 是 int 的子类，但 type 不认）


# =========================
# 2. isinstance()
# =========================
# 判断对象是否是某个类型或其子类的实例
# ✔️ 推荐用于类型判断

print(isinstance(a, bool))  # True
print(isinstance(a, int))   # True（bool 是 int 的子类）

# 支持多个类型
print(isinstance(a, (int, str)))  # True


# =========================
# 3. issubclass()
# =========================
# 判断一个类是否是另一个类的子类

print(issubclass(bool, int))   # True
print(issubclass(int, bool))   # False


# =========================
# 4. del
# =========================
# 删除变量名（引用），不是立刻销毁对象
# 对象是否销毁取决于是否还有引用

x = [1, 2, 3]
y = x
del x
print(y)   # [1, 2, 3]（对象还在）

# 删除多个变量
del y


# =========================
# 5. id()
# =========================
# 返回对象在内存中的唯一标识（CPython 中通常是内存地址）

a = 100
b = 100
print(id(a) == id(b))   # True（小整数缓存）


# =========================
# 6. dir()
# =========================
# 返回对象可用的属性和方法列表（调试 / 学习神器）

print(dir(a))


# =========================
# 7. hasattr / getattr / setattr / delattr
# =========================
# 反射相关：对对象属性进行“动态操作”

class A:
    x = 10

obj = A()

print(hasattr(obj, 'x'))        # True
print(getattr(obj, 'x'))        # 10

setattr(obj, 'y', 20)
print(obj.y)                    # 20

delattr(A, 'x')                 # 注意：删的是类属性
# print(obj.x)                  # AttributeError


# =========================
# 8. callable()
# =========================
# 判断对象是否可被调用（是否能 obj()）

def f():
    pass

print(callable(f))      # True
print(callable(a))      # False


# =========================
# 9. len()
# =========================
# 返回对象长度，本质调用 __len__()

lst = [1, 2, 3]
print(len(lst))         # 3


# =========================
# 10. help()
# =========================
# 查看对象 / 函数 / 类的文档说明

# help(list)
# help(isinstance)


# =========================
# 11. isinstance vs type（重点对比）
# =========================
# ✔️ 推荐使用 isinstance
# ❌ 不推荐用 type(x) == T 来判断类型

class Base: pass
class Child(Base): pass

c = Child()

print(type(c) == Base)          # False
print(isinstance(c, Base))      # True


# =========================
# 12. 常见类型判断组合（实战常用）
# =========================
def func(x):
    if isinstance(x, (int, float)):
        print("数字")
    elif isinstance(x, str):
        print("字符串")
    elif isinstance(x, (list, tuple, set)):
        print("容器类型")
    else:
        print("其他类型")


# =========================
# 一句话总结
# =========================
# type()        → 看“它本质是什么”
# isinstance() → 看“能不能当成某种类型用”
# issubclass() → 类与类之间的继承关系
# del           → 删除引用，不一定销毁对象
```

# 2、关键字

```python
# =========================
# Python 关键字（Keywords）速查笔记
# =========================
# 关键字是 Python 语法的一部分
# 不能作为变量名、函数名、类名使用
# 关键字数量会随 Python 版本略有变化（以下基于 Python 3.10+）


# =========================
# 1、逻辑 / 布尔相关
# =========================
# True      真
# False     假
# and       逻辑与
# or        逻辑或
# not       逻辑非

a = True
b = False
print(a and b)
print(a or b)
print(not a)


# =========================
# 2、条件控制
# =========================
# if        条件判断
# elif      否则如果
# else      否则

if a:
    print("yes")
elif b:
    print("maybe")
else:
    print("no")


# =========================
# 3、循环控制
# =========================
# for       for 循环
# while     while 循环
# break     跳出循环
# continue  跳过本次循环
# else      循环正常结束时执行（无 break）

for i in range(3):
    if i == 1:
        continue
    print(i)
else:
    print("循环结束")


# =========================
# 4、异常处理
# =========================
# try       尝试执行
# except    捕获异常
# else      无异常时执行
# finally   无论如何都会执行
# raise     主动抛出异常

try:
    x = 1 / 0
except ZeroDivisionError:
    print("除零错误")
finally:
    print("结束")


# =========================
# 5、函数相关
# =========================
# def       定义函数
# return    返回值
# lambda    匿名函数
# yield     生成器返回

def add(a, b):
    return a + b

f = lambda x: x * 2

def gen():
    yield 1
    yield 2


# =========================
# 6、类 / 面向对象
# =========================
# class     定义类
# self      实例自身（约定写法）
# super     调用父类
# pass      空语句，占位

class A:
    def __init__(self):
        pass


# =========================
# 7、模块与作用域
# =========================
# import    导入模块
# from      从模块导入
# as        别名
# global    声明全局变量
# nonlocal  声明外层函数变量

import math as m
from math import pi

x = 10
def f():
    global x
    x = 20


# =========================
# 8、数据与对象
# =========================
# None      空值 / 空对象
# del       删除引用
# is        身份判断
# in        成员判断

a = None
lst = [1, 2, 3]

print(a is None)
print(1 in lst)


# =========================
# 9、异步 / 上下文（进阶）
# =========================
# with      上下文管理
# async     异步定义
# await     等待异步结果

with open("test.txt", "w") as f:
    f.write("hello")


# =========================
# 10、模式匹配（Python 3.10+）
# =========================
# match     模式匹配
# case      分支匹配

x = 1
match x:
    case 1:
        print("one")
    case _:
        print("other")


# =========================
# 11、保留关键字（很少直接用）
# =========================
# assert    断言（调试用）
# finally   异常收尾
# yield     生成器
# await     异步等待


# =========================
# 12、查看当前 Python 版本关键字（推荐）
# =========================
import keyword
print(keyword.kwlist)


# =========================
# 总结（记忆建议）
# =========================
# 1️⃣ 控制流程：if / for / while / break / continue
# 2️⃣ 函数相关：def / return / lambda / yield
# 3️⃣ 异常处理：try / except / finally / raise
# 4️⃣ 面向对象：class / self / super / pass
# 5️⃣ 判断相关：is / in / None / True / False
```
