# 1、面向对象介绍

**面向对象**：用“类”来描述现实世界的事物，类是模板，对象是类的实例。

**类 (Class)**：定义属性和方法的模板。

**对象 (Object)**：类的实例。

**实例化**：由类创建对象的过程。

**方法 (Method)**：定义在类中的函数。

**属性**：对象或类中保存的数据。

**类变量**：类变量在整个实例化的对象中是公用的。类变量定义在类中且在函数体之外。类变量通常不作为实例变量使用。

**局部变量**：定义在方法中的变量，只作用于当前实例的类。

**实例变量**：在类的声明中，属性是用变量来表示的，这种变量就称为实例变量，实例变量就是一个用 self 修饰的变量。

**继承**：即一个派生类（derived class）继承基类（base class）的字段和方法。继承也允许把一个派生类的对象作为一个基类对象对待。例如，有这样一个设计：一个Dog类型的对象派生自Animal类，这是模拟"是一个（is-a）"关系（例图，Dog是一个Animal）

**方法重写**：如果从父类继承的方法不能满足子类的需求，可以对其进行改写，这个过程叫方法的覆盖（override），也称为方法的重写。

---

**Python面向对象特点**

- 不需要显式声明成员变量
- 通过 self 绑定实例属性
- 支持多继承
- 支持动态添加属性
- 一切皆对象（函数、类本身也是对象）

# 3、类变量 / 实例变量

```python
class MyClass:
    """一个简单的类实例"""

    # 类变量（所有实例共享）
    i = 12345

    def __init__(self):
        # 实例变量（每个实例独立）
        self.data = []

    def f(self):
        return 'hello world'
```

# 4、类的实例化与访问

```python
x = MyClass()   # 实例化

print("类变量 i：", x.i)
print("实例变量 data：", x.data)
print("方法 f：", x.f())
```

# 6、构造方法 `__init__`

```python
# __init__ 在对象创建时自动调用
# 用于初始化实例变量

class People:
    name = ''
    age = 0
    __weight = 0  # 私有属性（外部不可直接访问）

    def __init__(self, n, a, w):
        self.name = n
        self.age = a
        self.__weight = w

    def speak(self):
        print("%s 说: 我 %d 岁。" % (self.name, self.age))


p = People('runoob', 10, 30)
p.speak()
```

# 7、私有属性 / 私有方法

```python
# 以 __ 开头的属性或方法是“伪私有”
# 实际会被重命名为 _类名__属性名

class Test:
    def __init__(self):
        self.__secret = 123

    def __private_method(self):
        print("私有方法")

    def show(self):
        print(self.__secret)
        self.__private_method()
```

# 8、继承（单继承）

```python
class Student(People):
    grade = ''

    def __init__(self, n, a, w, g):
        # 调用父类构造方法
        People.__init__(self, n, a, w)
        self.grade = g

    # 方法重写（override）
    def speak(self):
        print("%s 说: 我 %d 岁了，我在读 %d 年级" %
              (self.name, self.age, self.grade))


s = Student('ken', 10, 60, 3)
s.speak()
```

# 9、多继承

```python
# Python 支持多继承
# 方法查找顺序：从左到右（MRO）

class Speaker:
    topic = ''
    name = ''

    def __init__(self, n, t):
        self.name = n
        self.topic = t

    def speak(self):
        print("我叫 %s，我演讲的主题是 %s" % (self.name, self.topic))


class Sample(Speaker, Student):
    def __init__(self, n, a, w, g, t):
        Student.__init__(self, n, a, w, g)
        Speaker.__init__(self, n, t)


test = Sample("Tim", 25, 80, 4, "Python")
test.speak()   # 默认调用 Speaker.speak（左侧优先）
```

# 10、方法重写 & super

```python
class Parent:
    def myMethod(self):
        print('调用父类方法')


class Child(Parent):
    def myMethod(self):
        print('调用子类方法')


c = Child()
c.myMethod()               # 子类方法
super(Child, c).myMethod() # 父类方法
```

# 12、常见魔术方法（特殊方法）

```python
# =========================
# Python 常用魔法方法（dander methods）
# =========================

# __init__      构造函数
# 对象创建时自动调用，用于初始化实例属性
class A:
    def __init__(self, x):
        self.x = x

a = A(10)


# __del__       析构函数
# 对象被垃圾回收时调用（不保证立即执行）
# 一般不建议依赖它做关键逻辑
class B:
    def __del__(self):
        print("对象被销毁")

b = B()
del b


# __repr__      对象字符串表示
# print(obj) / 直接在解释器中输入 obj 时调用
class C:
    def __repr__(self):
        return "这是一个 C 对象"

print(C())


# __len__       len(obj)
# 使对象支持 len()
class D:
    def __len__(self):
        return 5

d = D()
print(len(d))   # 5


# __getitem__   obj[key]
# 使对象支持下标读取
class E:
    def __init__(self):
        self.data = [1, 2, 3]

    def __getitem__(self, index):
        return self.data[index]

e = E()
print(e[1])     # 2


# __setitem__   obj[key] = value
# 使对象支持下标赋值
class F:
    def __init__(self):
        self.data = {}

    def __setitem__(self, key, value):
        self.data[key] = value

f = F()
f["a"] = 100
print(f.data)   # {'a': 100}


# __call__      obj()
# 使对象像函数一样被调用
class G:
    def __call__(self, x):
        return x * 2

g = G()
print(g(5))     # 10


# __add__       +
# 定义对象之间的加法行为
class H:
    def __init__(self, x):
        self.x = x

    def __add__(self, other):
        return self.x + other.x

print(H(3) + H(4))   # 7


# __sub__       -
# 定义对象之间的减法行为
class I:
    def __init__(self, x):
        self.x = x

    def __sub__(self, other):
        return self.x - other.x

print(I(10) - I(3))  # 7


# __mul__       *
# 定义对象之间的乘法行为
class J:
    def __init__(self, x):
        self.x = x

    def __mul__(self, other):
        return self.x * other.x

print(J(2) * J(4))   # 8


# __truediv__   /
# 定义对象之间的除法行为
class K:
    def __init__(self, x):
        self.x = x

    def __truediv__(self, other):
        return self.x / other.x

print(K(8) / K(2))   # 4.0


# __mod__       %
# 定义对象之间的取模行为
class L:
    def __init__(self, x):
        self.x = x

    def __mod__(self, other):
        return self.x % other.x

print(L(10) % L(3))  # 1


# __pow__       **
# 定义对象之间的幂运算行为
class M:
    def __init__(self, x):
        self.x = x

    def __pow__(self, other):
        return self.x ** other.x

print(M(2) ** M(3))  # 8
```
