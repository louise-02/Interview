# 1、函数定义

```python
# =========================
# 1. 函数定义
# =========================
# 使用 def 关键字定义函数
# 函数可以有参数，也可以没有
# 函数体通过缩进表示

def add(a, b):
    return a + b

print(add(1, 2))   # 3
```

# 2、参数传递机制（对象引用）

```python
# =========================
# 2. 参数传递机制（对象引用）
# =========================
# Python 采用 对象引用传递
# 变量没有类型，类型属于对象


# ---- 不可变类型示例（int / str / tuple 等）----
def f(x):
    x = 100

a = 10
f(a)
print(a)           # 10（不受影响）


# ---- 可变类型示例（list / dict / set）----
def g(lst):
    lst.append(4)

b = [1, 2, 3]
g(b)
print(b)           # [1, 2, 3, 4]
```

# 3、参数类型

```python
# =========================
# 3. 参数类型
# =========================

# ---- 3.1 必需参数 ----
def print_info(name, age):
    print(name, age)

print_info("Tom", 18)


# ---- 3.2 关键字参数 ----
def print_info(name, age):
    print(name, age)

print_info(age=18, name="Tom")


# ---- 3.3 默认参数 ----
def print_info(name, age=18):
    print(name, age)

print_info("Tom")
print_info("Tom", 20)


# ⚠️ 默认参数不要使用可变类型
def bad_func(a=[]):
    a.append(1)
    return a
```

# 4、不定长参数

```python
# =========================
# 4. 不定长参数
# =========================

# ---- *args：以元组形式接收 ----
def func(*args):
    print(args)
    print(type(args))

func(1, 2, 3)
# (1, 2, 3)
# <class 'tuple'>


# ---- **kwargs：以字典形式接收 ----
def func(**kwargs):
    print(kwargs)
    print(type(kwargs))

func(a=1, b=2)
# {'a': 1, 'b': 2}
# <class 'dict'>
```

# 5、return 语句

```python
# =========================
# 5. return 语句
# =========================
# return 用于结束函数并返回值
# 没有 return 或 return 后无值，默认返回 None

def f1():
    return 1

def f2():
    pass

print(f1())        # 1
print(f2())        # None
```

# 6、lambda 匿名函数

```python
# =========================
# 6. lambda 匿名函数
# =========================
# lambda 只能写一行表达式
# 适合临时、简单逻辑

add = lambda a, b: a + b
print(add(1, 2))   # 3


# 常见用法（配合 sorted / map / filter）
lst = [(1, 3), (4, 1), (2, 2)]
lst.sort(key=lambda x: x[1])
print(lst)
# [(4, 1), (2, 2), (1, 3)]
```

# 7、强制位置参数 / 关键字参数（Python 3.8+）

```python
# =========================
# 7. 强制位置参数 / 关键字参数（Python 3.8+）
# =========================
# / 之前的参数只能用位置传参
# * 之后的参数只能用关键字传参

def f(a, b, /, c, d, *, e, f):
    print(a, b, c, d, e, f)

f(10, 20, 30, d=40, e=50, f=60)
# 正确

# f(10, b=20, c=30, d=40, e=50, f=60)
# ❌ b 不能使用关键字
```

# 8、全局变量 global

```python
# =========================
# 8. 全局变量 global
# =========================
# 在函数中修改全局变量必须使用 global

a = 10

def update_global():
    global a
    a = 5

update_global()
print(a)           # 5
```
