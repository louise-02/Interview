# 1、输出 print + format

```python
# =========================
# 1. print 基础用法
# =========================

print("Hello World")

# print 可输出多个值，默认用空格分隔
print("a", "b", "c")              # a b c

# print 会自动在末尾加换行
print("line1")
print("line2")


# =========================
# 2. end 参数（控制结尾）
# =========================
# 默认：end = "\n"

print("Hello", end="")            # 不换行
print("World")                    # HelloWorld

print("A", end="--")
print("B")                        # A--B

# 常用于读取文件时防止重复换行
# print(line, end="")


# =========================
# 3. sep 参数（分隔符）
# =========================
# 默认：sep = " "

print(1, 2, 3)                    # 1 2 3
print(1, 2, 3, sep=",")           # 1,2,3
print("2025", "01", "01", sep="-")# 2025-01-01


# =========================
# 4. format 基础格式化
# =========================

print('站点列表 {0}, {1}, 和 {2}'.format(
    'Google', 'Runoob', 'Taobao'
))

# 按索引复用
print('{0} {0} {1}'.format('hello', 'world'))
# hello hello world


# =========================
# 5. format 命名参数
# =========================

print('站点列表 {a}, {b}, 和 {c}'.format(
    a='Google', b='Runoob', c='Taobao'
))

print('{name}网址：{site}'.format(
    name='菜鸟教程',
    site='www.runoob.com'
))


# =========================
# 6. 混合使用（位置 + 命名）
# =========================

print('{0} {name}'.format(
    'Hello',
    name='World'
))


# =========================
# 7. 控制浮点数精度
# =========================

import math

# 保留 3 位小数
print('PI ≈ {0:.3f}'.format(math.pi))

# 保留 2 位小数
print('{:.2f}'.format(3.14159))

# 宽度 + 精度
print('{:8.2f}'.format(3.14159))
#    3.14


# =========================
# 8. 整数格式化
# =========================

print('{:d}'.format(10))          # 十进制
print('{:b}'.format(10))          # 二进制
print('{:o}'.format(10))          # 八进制
print('{:x}'.format(255))         # 十六进制
print('{:X}'.format(255))         # 十六进制（大写）


# =========================
# 9. 对齐方式
# =========================

print('|{:10}|'.format('left'))   # 默认左对齐
print('|{:>10}|'.format('right')) # 右对齐
print('|{:^10}|'.format('mid'))   # 居中对齐

# 填充字符
print('{:*^10}'.format('hi'))     # ****hi****


# =========================
# 10. 百分比显示
# =========================

print('{:.2%}'.format(0.1234))    # 12.34%


# =========================
# 11. f-string（Python 3.6+ 推荐）
# =========================

name = "Tom"
age = 18

print(f"name={name}, age={age}")

# 表达式
print(f"2 + 3 = {2 + 3}")

# 控制精度
print(f"PI ≈ {math.pi:.3f}")


# =========================
# 12. print 到文件
# =========================

with open("out.txt", "w") as f:
    print("写入文件的内容", file=f)
    
    
# =========================
# 13. 常见使用场景
# =========================

# 防止多余换行
print(line, end="")

# 日志式输出
print(f"[INFO] user={name}, age={age}")

# 表格输出
for i in range(3):
    print('{:<5}{:<5}'.format(i, i*i))
```

# 3、open 函数

```python
# =========================
# 1 open 函数
# =========================
# open(filename, mode)

# r   只读（默认）
# rb  二进制只读
# r+  读写
# w   写（清空）
# wb  二进制写
# w+  读写（清空）
# a   追加
# ab  二进制追加
# a+  读写追加

# =========================
# 2 文件对象常用方法
# =========================

f = open("test.txt", "w+")
f.write("hello\n")
f.flush()
f.seek(0)

print(f.read())        # 读取全部
print(f.tell())        # 当前指针位置

f.close()

# f.read(size)      读取指定大小
# f.readline()      读取一行
# f.readlines()     读取所有行
# f.write(str)      写入字符串
# f.tell()          获取指针位置
# f.flush()         刷新缓冲区
# f.close()         关闭文件

# =========================
# readline() 循环读取完整文件（方式一）
# =========================
# 最直观写法（不推荐，偏底层）

with open("test.txt", "r") as f:
    while True:
        line = f.readline()
        if not line:          # 读到文件结尾，返回 ''
            break
        print(line, end="")


# =========================
# readline() 循环读取完整文件（方式二）
# =========================
# 使用 for（推荐）
# 本质：for 内部一直在调用 readline()

with open("test.txt", "r") as f:
    for line in f:
        print(line, end="")
        
        
# =========================
# 不同系统的换行符
# =========================
# Linux / macOS / Python 内部标准
#   \n
#
# Windows 文本文件传统格式
#   \r\n

# =========================
# Python 的“通用换行”机制
# =========================
# 1️⃣ 写文件时：
#    你写 '\n' 即可
#    Python 在 Windows 上会自动转换为 '\r\n'
#
# 2️⃣ 读文件时：
#    不管文件里是 '\n' 还是 '\r\n'
#    Python 都会统一解析成 '\n'

# =========================
# 推荐写法（跨平台标准）
# =========================
# 永远只写 '\n'
# 永远用文本模式 'r' / 'w'
```

# 4、pickle 模块（序列化）

```python
# =========================
# pickle 模块（序列化）
# =========================
# pickle 用于 Python 对象 -> 文件
# pickle 只能用于 Python


# ---- 写入 ----
import pickle

data1 = {
    'a': [1, 2.0, 3, 4+6j],
    'b': ('string', 'Unicode string'),
    'c': None
}

selfref_list = [1, 2, 3]
selfref_list.append(selfref_list)

output = open('data.pkl', 'wb')
pickle.dump(data1, output)
pickle.dump(selfref_list, output, -1)
output.close()


# ---- 读取 ----
import pprint

pkl_file = open('data.pkl', 'rb')

data1 = pickle.load(pkl_file)
pprint.pprint(data1)

data2 = pickle.load(pkl_file)
pprint.pprint(data2)

pkl_file.close()
```

# 5、os 模块

```python
# =========================
# os 模块
# =========================

import os

# 创建目录
os.mkdir("test_dir")

# 删除文件
os.remove("test.txt")

# 删除空目录
os.rmdir("test_dir")

# 修改权限
os.chmod("test.txt", 0o777)

# 遍历目录
for root, dirs, files in os.walk("."):
    print(root, dirs, files)

# 执行系统命令（无输出）
os.system("ls")

# 执行系统命令（有输出）
result = os.popen("ls").read()
print(result)


# =========================
# subprocess（更强的命令执行）
# =========================
import subprocess

p = subprocess.Popen(
    "ls",
    shell=True,
    stdout=subprocess.PIPE
)

print(p.stdout.read().decode())
```
