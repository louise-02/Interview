# 1、相关概念

散列表（Hash table，也叫哈希表），是根据关键码值(Key value)而直接进行访问的数据结构。也就是说，它通过把关键码值映射到表中一个位置来访问记录，以加快查找的速度。这个映射函数叫做散列函数，存放记录的数组叫做散列表。

通常用 数组+链表 或 数组+二叉树。

```
哈希表示例：

索引    数据
[0]     
[1]     
[2]     12
[3]     33
[4]     
[5]     5 → 25
[6]     
[7]     
[8]     8
[9]     19
[10]    
[11]    
[12]    
[13]    
[14]    
[15]    

哈希函数：h(key) = key % 16

插入过程：
5  → 5%16=5   → [5]
12 → 12%16=12 → [12]
25 → 25%16=9  → [9]
8  → 8%16=8   → [8]
19 → 19%16=3  → [3]
33 → 33%16=1  → [1]
```

# 2、代码实现

```java
public class HashTable {

    EmpLinkList[] empLinkLists;

    int size;

    public static void main(String[] args) {
        HashTable hashTable = new HashTable(8);
        Emp emp = new Emp();
        emp.setId(1);
        emp.setName("pick");
        hashTable.put(emp);
        Emp emp2 = new Emp();
        emp2.setId(9);
        emp2.setName("pick2");
        hashTable.put(emp2);
        Emp emp3 = new Emp();
        emp3.setId(2);
        emp3.setName("pick3");
        hashTable.put(emp3);
        hashTable.list();
        System.out.println(hashTable.findById(1));
    }

    public HashTable(int size) {
        this.empLinkLists = new EmpLinkList[size];
        this.size = size;
    }

    public Emp findById(int id){
        int hash = getHash(id);
        if (empLinkLists[hash] == null){
            return null;
        }
        Emp head = empLinkLists[hash].head;
        while (head != null){
            if (head.id == id){
                return head;
            }
            head = head.next;
        }
        return null;
    }

    public void put(Emp emp) {
        int hash = getHash(emp.getId());
        if (empLinkLists[hash] == null) {
            empLinkLists[hash] = new EmpLinkList();
        }
        Emp head = empLinkLists[hash].head;
        if (head == null) {
            empLinkLists[hash].head = emp;
            return;
        }
        while (true) {
            if (head.next == null) {
                head.next = emp;
                return;
            }
            head = head.next;
        }
    }

    public void list() {
        for (int i = 0; i < empLinkLists.length; i++) {
            System.out.println("第" + i + "条");
            if (empLinkLists[i] == null){
                System.out.print("null");
            }else {
                empLinkLists[i].list();
            }
            System.out.println();
        }
    }

    private int getHash(int id) {
        return id % size;
    }
}

class EmpLinkList {
    Emp head;

    public void list() {
        Emp tmp = head;
        while (tmp != null) {
            System.out.print(tmp + " ");
            tmp = tmp.next;
        }
    }
}

class Emp {
    int id;
    String name;
    Emp next;

    public int getId() {
        return id;
    }

    public void setId(int id) {
        this.id = id;
    }

    public String getName() {
        return name;
    }

    public void setName(String name) {
        this.name = name;
    }

    @Override
    public String toString() {
        return "Emp{" +
                "id=" + id +
                ", name='" + name + '\'' +
                '}';
    }
}
```
